# Kafka consumer → Batch inference → Redis pub/sub

def consume_and_predict(consumer, model, batch_size=32):
    batch = []
    for message in consumer:
        batch.append(message.value)
        
        if len(batch) >= batch_size:
            # Tokenize batch efficiently
            inputs = tokenizer(batch, padding=True,
                             truncation=True,
                             return_tensors='tf',
                             max_length=128)
            
            # Batch prediction — 8ms for 32 tweets
            predictions = model(inputs)['logits']
            labels = tf.argmax(predictions, axis=-1).numpy()
            
            # Publish to Redis for dashboard
            for tweet, label in zip(batch, labels):
                redis_client.publish('sentiment_stream',
                    json.dumps({"text": tweet, 
                                "label": id2label[label]}))
            batch = []
