class VQAModel(tf.keras.Model):
    def __init__(self, num_answers=3129):
        super().__init__()
        # Vision encoder — pretrained ResNet-152
        self.vision_encoder = tf.keras.applications.ResNet152(
            weights='imagenet', include_top=False,
            pooling='avg')
        self.vision_proj = tf.keras.layers.Dense(768)
        
        # Text encoder — pretrained BERT-base
        self.text_encoder = TFBertModel.from_pretrained(
            'bert-base-uncased')
        
        # Multimodal fusion
        self.attention = MultiHeadAttention(num_heads=8, key_dim=64)
        self.classifier = tf.keras.layers.Dense(num_answers, 'softmax')

    def call(self, image, input_ids, attention_mask):
        v = self.vision_proj(self.vision_encoder(image))
        t = self.text_encoder(input_ids, attention_mask=attention_mask).last_hidden_state
        fused = self.attention(t, v)
        return self.classifier(fused[:, 0, :])
